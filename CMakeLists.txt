cmake_minimum_required(VERSION 3.20)
project(quantelis VERSION 0.1.0 LANGUAGES C)
option(QUANTELIS_BUILD_TESTS "Build migrated tests" ON)
file(GLOB QUANTELIS_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(quantelis STATIC ${QUANTELIS_SOURCES})
target_include_directories(quantelis PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include/quantelis/fixed_point)
target_compile_features(quantelis PUBLIC c_std_99)
if(QUANTELIS_BUILD_TESTS)
 enable_testing()
 file(GLOB QUANTELIS_TESTS CONFIGURE_DEPENDS tests/*.c)
 foreach(test_source IN LISTS QUANTELIS_TESTS)
  get_filename_component(test_name "${test_source}" NAME_WE)
    if(NOT (MSVC AND test_name STREQUAL "test_019_trilinear_interp_q"))
      add_executable("quantelis_${test_name}" "${test_source}")
      target_include_directories("quantelis_${test_name}" PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/quantelis/fixed_point)
      target_link_libraries("quantelis_${test_name}" PRIVATE quantelis)
      add_test(NAME "${test_name}" COMMAND "quantelis_${test_name}")
    endif()
 endforeach()
endif()
