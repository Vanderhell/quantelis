cmake_minimum_required(VERSION 3.20)
project(quantelis VERSION 0.1.0 LANGUAGES C)

option(QUANTELIS_BUILD_TESTS "Build Quantelis tests" ON)
option(QUANTELIS_STRICT_WARNINGS "Enable strict compiler warnings" OFF)

file(GLOB QUANTELIS_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c")
add_library(quantelis STATIC ${QUANTELIS_SOURCES})
target_include_directories(quantelis PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/quantelis/fixed_point>"
)
target_compile_features(quantelis PUBLIC c_std_99)

if(QUANTELIS_STRICT_WARNINGS)
    if(MSVC)
        target_compile_options(quantelis PRIVATE /W4 /WX)
    else()
        target_compile_options(quantelis PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()

if(QUANTELIS_BUILD_TESTS)
    enable_testing()
    file(GLOB QUANTELIS_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.c")
    foreach(test_source IN LISTS QUANTELIS_TESTS)
        get_filename_component(test_name "${test_source}" NAME_WE)
        add_executable("quantelis_${test_name}" "${test_source}")
        target_link_libraries("quantelis_${test_name}" PRIVATE quantelis)
        if(UNIX)
            target_link_libraries("quantelis_${test_name}" PRIVATE m)
        endif()
        if(QUANTELIS_STRICT_WARNINGS)
            if(MSVC)
                target_compile_options("quantelis_${test_name}" PRIVATE /W4 /WX)
            else()
                target_compile_options("quantelis_${test_name}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
            endif()
        endif()
        add_test(NAME "${test_name}" COMMAND "quantelis_${test_name}")
    endforeach()
endif()
